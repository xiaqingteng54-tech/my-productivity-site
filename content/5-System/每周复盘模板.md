# 周复盘 · <% tp.date.now("YYYY-[W]WW") %>

> 周期：`<% tp.date.weekday("YYYY-MM-DD", 1) %>` → `<% tp.date.weekday("YYYY-MM-DD", 7) %>`

## 1. 本周完成的任务

```dataview
TASK
FROM "0-Inbox" OR "1-Projects" OR "2-Areas" OR "3-Resources"
WHERE completed
  AND completion
  AND completion >= date(this.week_start)
  AND completion <= date(this.week_end)
GROUP BY file.link
```

## 2. 本周新建的笔记

```dataview
TABLE WITHOUT ID
  file.link AS "笔记",
  file.folder AS "所在目录",
  dateformat(file.ctime, "MM-dd HH:mm") AS "创建时间"
FROM "0-Inbox" OR "1-Projects" OR "2-Areas" OR "3-Resources"
WHERE file.ctime >= date(this.week_start)
  AND file.ctime <= date(this.week_end) + dur(1 day)
SORT file.ctime DESC
```

## 3. 本周活跃项目

TABLE WITHOUT ID

## 4. 本周数据概览

```dataviewjs
const s = dv.date(dv.current().week_start);
const e = dv.date(dv.current().week_end);

const pages = dv.pages('"0-Inbox" OR "1-Projects" OR "2-Areas" OR "3-Resources"');

const created = pages.where(p => p.file.ctime >= s && p.file.ctime <= e.plus({ days: 1 })).length;
const touched = pages.where(p => p.file.mtime >= s).length;

const tasks = pages.file.tasks;
const done  = tasks.where(t => t.completed && t.completion && t.completion >= s && t.completion <= e).length;
const open  = tasks.where(t => !t.completed).length;
const rate  = (done + open) === 0 ? 0 : Math.round(done / (done + open) * 100);

dv.table(
  ["指标", "本周"],
  [
    ["新建笔记", created],
    ["有改动笔记", touched],
    ["完成任务", done],
    ["当前未完成", open],
    ["任务完成率", rate + "%"],
  ]
);
```
